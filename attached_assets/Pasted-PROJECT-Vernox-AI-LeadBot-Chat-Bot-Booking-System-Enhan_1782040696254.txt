export const parseBookingDatetime = (rawString: string): Date | null => {
  // Try ISO format first
  const isoDate = new Date(rawString);
  if (!isNaN(isoDate.getTime())) return isoDate;
  
  // Try common natural language patterns
  const patterns = [
    { regex: /tomorrow\s+at\s+(\d{1,2}):?(\d{2})?\s*(am|pm)?/i, fn: parseTomorrowTime },
    { regex: /(\d{1,2})\s+(january|february|march|april|may|june|july|august|september|october|november|december)\s+at\s+(\d{1,2}):?(\d{2})?\s*(am|pm)?/i, fn: parseSpecificDate },
    { regex: /next\s+(monday|tuesday|wednesday|thursday|friday|saturday|sunday)\s+at\s+(\d{1,2}):?(\d{2})?\s*(am|pm)?/i, fn: parseNextDayOfWeek },
  ];
  
  for (const pattern of patterns) {
    const match = rawString.match(pattern.regex);
    if (match) {
      const parsed = pattern.fn(match);
      if (parsed) return parsed;
    }
  }
  
  // Fallback: tomorrow at 9 AM UTC
  const fallback = new Date();
  fallback.setDate(fallback.getDate() + 1);
  fallback.setHours(9, 0, 0, 0);
  
  logger.warn({
    msg: "Booking datetime unparseable, using fallback",
    rawString,
    fallbackDate: fallback.toISOString()
  });
  
  return fallback;
};
