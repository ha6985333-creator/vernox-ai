You are adding email confirmations to Vernox AI. Right now when someone books, they 
get nothing — no confirmation, no reminder, nothing. This needs to change immediately.

CURRENT STATE:
- Backend: Node.js/Express
- Email service: Resend (already integrated)
- Database: PostgreSQL
- Booking flow: Complete (creates calendar event + lead record)

THE PROBLEM:
When a customer books an appointment:
1. Google Calendar gets updated ✓
2. Lead record is created ✓
3. But the customer receives NO email confirmation
4. They don't remember they booked
5. No-show rate goes up 40-60%

YOUR TASK:
Build an automated email confirmation system that:
1. Sends confirmation email immediately after booking
2. Includes appointment details (date, time, location)
3. Includes business info (name, phone, address)
4. Includes links to reschedule or cancel
5. Uses professional HTML email template
6. Tracks if email was delivered

IMPLEMENTATION REQUIREMENTS:

### STEP 1: Create Email Templates Table

```sql
CREATE TABLE email_templates (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  business_id UUID NOT NULL REFERENCES businesses(id) ON DELETE CASCADE,
  template_type VARCHAR(100), -- 'booking_confirmation', 'reminder', 'cancellation', etc.
  subject VARCHAR(255),
  html_body TEXT,
  text_body TEXT,
  variables TEXT[], -- ['customer_name', 'appointment_time', 'business_name']
  is_default BOOLEAN DEFAULT false,
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE email_logs (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  lead_id UUID NOT NULL REFERENCES leads(id),
  business_id UUID NOT NULL REFERENCES businesses(id),
  template_type VARCHAR(100),
  recipient_email VARCHAR(255),
  subject VARCHAR(255),
  status VARCHAR(50), -- 'pending', 'sent', 'delivered', 'bounced', 'failed'
  resend_message_id VARCHAR(255),
  sent_at TIMESTAMP,
  delivered_at TIMESTAMP,
  error_message TEXT,
  created_at TIMESTAMP DEFAULT NOW()
);
```

### STEP 3: Email Service Module

```javascript
// backend/services/emailService.js

const { Resend } = require('resend');

const resend = new Resend(process.env.RESEND_API_KEY);

async function sendBookingConfirmation(leadId, businessId, customerData, appointmentData) {
  try {
    // 1. Get business info
    const business = await db.query(
      'SELECT business_name, business_email, business_phone, business_address FROM businesses WHERE id = $1',
      [businessId]
    );
    
    // 2. Get or create email template
    let template = await db.query(
      'SELECT * FROM email_templates WHERE business_id = $1 AND template_type = $2',
      [businessId, 'booking_confirmation']
    );
    
    if (!template.rows[0]) {
      // Use default template
      template = defaultTemplates.booking_confirmation;
    } else {
      template = template.rows[0];
    }
    
    // 3. Replace variables in template
    const variables = {
      '{customer_name}': customerData.name,
      '{business_name}': business.rows[0].business_name,
      '{appointment_date}': new Date(appointmentData.datetime).toLocaleDateString(),
      '{appointment_time}': new Date(appointmentData.datetime).toLocaleTimeString([], 
        { hour: '2-digit', minute: '2-digit' }),
      '{service}': customerData.service,
      '{business_address}': business.rows[0].business_address,
      '{business_phone}': business.rows[0].business_phone,
      '{reschedule_link}': `https://vernoxai.digital/reschedule?leadId=${leadId}&token=xxx`
    };
    
    let htmlBody = template.htmlBody || template.html_body;
    let subject = template.subject;
    
    Object.keys(variables).forEach(key => {
      htmlBody = htmlBody.replace(key, variables[key]);
      subject = subject.replace(key, variables[key]);
    });
    
    // 4. Send via Resend
    const response = await resend.emails.send({
      from: `${business.rows[0].business_name} <noreply@vernoxai.digital>`,
      to: customerData.email,
      subject: subject,
      html: htmlBody,
      replyTo: business.rows[0].business_email || 'support@vernoxai.digital'
    });
    
    // 5. Log the email
    await db.query(
      `INSERT INTO email_logs 
       (lead_id, business_id, template_type, recipient_email, subject, status, resend_message_id, sent_at)
       VALUES ($1, $2, $3, $4, $5, $6, $7, NOW())`,
      [leadId, businessId, 'booking_confirmation', customerData.email, subject, 'sent', response.id]
    );
    
    return { success: true, messageId: response.id };
  } catch (error) {
    console.error('Email send error:', error);
    
    // Log failure
    await db.query(
      `INSERT INTO email_logs 
       (lead_id, business_id, template_type, recipient_email, status, error_message)
       VALUES ($1, $2, $3, $4, $5, $6)`,
      [leadId, businessId, 'booking_confirmation', customerData.email, 'failed', error.message]
    );
    
    return { success: false, error: error.message };
  }
}

module.exports = { sendBookingConfirmation };
```

// backend/routes/emailLogs.js

app.get('/api/email-logs/:businessId', async (req, res) => {
  const logs = await db.query(
    `SELECT el.*, l.customer_name, l.customer_email
     FROM email_logs el
     JOIN leads l ON el.lead_id = l.id
     WHERE el.business_id = $1
     ORDER BY el.created_at DESC
     LIMIT 50`,
    [req.params.businessId]
  );
  
  res.json(logs.rows);
});

app.get('/api/email-stats/:businessId', async (req, res) => {
  const stats = await db.query(`
    SELECT
      COUNT(*) as total_sent,
      SUM(CASE WHEN status = 'sent' THEN 1 ELSE 0 END) as sent,
      SUM(CASE WHEN status = 'delivered' THEN 1 ELSE 0 END) as delivered,
      SUM(CASE WHEN status = 'bounced' THEN 1 ELSE 0 END) as bounced,
      SUM(CASE WHEN status = 'failed' THEN 1 ELSE 0 END) as failed
    FROM email_logs
    WHERE business_id = $1
  `, [req.params.businessId]);
  
  res.json(stats.rows[0]);
});
