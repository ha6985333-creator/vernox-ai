You are building a simplified pricing tier system for Vernox AI. There is NO Stripe.
Instead: Customer selects a tier → fills a form → you manually send them a payment link.

CURRENT STATE:
- Backend: Node.js/Express
- Database: PostgreSQL via Supabase
- Payment: Manual (you send payment links via email)
- Frontend: React
- Already have: Form that collects name, email, business name, phone

THE PROBLEM:
Right now everyone sees $899/month and gets the same offer. You need:
- Three pricing tiers (Pro $899, Business $1,999, Enterprise $4,999)
- Customer selects tier
- Fills form with their details
- You get notified
- Features lock/unlock based on tier they select

YOUR TASK:
Build a pricing tier system where:
1. Pricing page shows three tiers with features
2. Customer clicks tier → form appears (pre-filled with tier info)
3. Form collects name, email, business name, phone, services offered
4. Submission creates "lead" + sends you admin notification
5. You manually follow up with payment link
6. Once paid, tier is activated in their account
7. Features unlock/lock based on tier

IMPLEMENTATION REQUIREMENTS:

### STEP 1: Create Subscription Tables (Simplified - No Stripe)

```sql
-- Subscription tiers definition
CREATE TABLE subscription_tiers (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name VARCHAR(100), -- 'Pro', 'Business', 'Enterprise'
  slug VARCHAR(50), -- 'pro', 'business', 'enterprise'
  price_monthly INTEGER, -- in cents (e.g., 89900 = $899)
  description TEXT,
  features JSONB, -- Features included in this tier
  display_order INTEGER,
  created_at TIMESTAMP DEFAULT NOW()
);

-- Customer subscriptions
CREATE TABLE subscriptions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  business_id UUID NOT NULL UNIQUE REFERENCES businesses(id) ON DELETE CASCADE,
  tier_id UUID NOT NULL REFERENCES subscription_tiers(id),
  
  -- Payment tracking (manual)
  payment_status VARCHAR(50) DEFAULT 'pending', -- 'pending', 'payment_sent', 'paid', 'canceled'
  payment_link_sent_at TIMESTAMP,
  paid_at TIMESTAMP,
  
  -- Dates
  subscription_start TIMESTAMP,
  subscription_end TIMESTAMP,
  next_renewal_date TIMESTAMP,
  
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW(),
  canceled_at TIMESTAMP
);

-- Signup leads (capture before payment)
CREATE TABLE signup_leads (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tier_id UUID NOT NULL REFERENCES subscription_tiers(id),
  
  -- Customer info from form
  customer_name VARCHAR(255) NOT NULL,
  customer_email VARCHAR(255) NOT NULL,
  business_name VARCHAR(255) NOT NULL,
  business_phone VARCHAR(20),
  business_services TEXT[], -- Services they offer
  
  -- Status
  status VARCHAR(50) DEFAULT 'new', -- 'new', 'payment_link_sent', 'paid', 'rejected'
  admin_notes TEXT,
  
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);

-- Payment sent history (track manual payments)
CREATE TABLE payment_links_sent (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  signup_lead_id UUID NOT NULL REFERENCES signup_leads(id),
  payment_method VARCHAR(50), -- 'payoneer', 'nsave', 'bank_transfer', etc.
  payment_link TEXT,
  sent_at TIMESTAMP DEFAULT NOW(),
  recipient_email VARCHAR(255),
  created_at TIMESTAMP DEFAULT NOW()
);
```

// backend/routes/signup.js

app.post('/api/signup/submit', async (req, res) => {
  const {
    tierId,
    name,
    email,
    businessName,
    businessPhone,
    businessServices
  } = req.body;
  
  try {
    // Validate tier
    const tier = await db.query(
      'SELECT id FROM subscription_tiers WHERE slug = $1',
      [tierId]
    );
    
    if (!tier.rows[0]) {
      return res.status(400).json({ error: 'Invalid tier' });
    }
    
    // Create signup lead
    const lead = await db.query(
      `INSERT INTO signup_leads 
       (tier_id, customer_name, customer_email, business_name, business_phone, business_services)
       VALUES ($1, $2, $3, $4, $5, $6)
       RETURNING id, tier_id`,
      [tier.rows[0].id, name, email, businessName, businessPhone, businessServices.split(',')]
    );
    
    // Send admin notification
    await sendAdminNotification(lead.rows[0], name, email, businessName, tierId);
    
    // Send customer confirmation email
    await sendCustomerConfirmationEmail(email, name, businessName, tierId);
    
    res.json({ 
      success: true, 
      leadId: lead.rows[0].id,
      message: 'Signup successful. Payment link coming soon.'
    });
    
  } catch (error) {
    console.error('Signup error:', error);
    res.status(500).json({ error: 'Signup failed' });
  }
});

// frontend/pages/AdminSignups.jsx

export default function AdminSignups() {
  const [signups, setSignups] = useState([]);
  const [filter, setFilter] = useState('new');
  
  useEffect(() => {
    fetchSignups();
  }, [filter]);
  
  const fetchSignups = async () => {
    const res = await fetch(`/api/admin/signups?status=${filter}`);
    const data = await res.json();
    setSignups(data);
  };
  
  const handleSendPaymentLink = async (leadId) => {
    const paymentLink = prompt('Paste payment link:');
    if (paymentLink) {
      await fetch(`/api/admin/signups/${leadId}/send-link`, {
        method: 'POST',
        body: JSON.stringify({ paymentLink })
      });
      fetchSignups();
    }
  };
  
  const handleMarkPaid = async (leadId) => {
    await fetch(`/api/admin/signups/${leadId}/mark-paid`, { method: 'POST' });
    fetchSignups();
  };
  
  return (
    <div className="admin-signups">
      <h2>New Signups ({signups.length})</h2>
      
      <table>
        <thead>
          <tr>
            <th>Date</th>
            <th>Name</th>
            <th>Business</th>
            <th>Tier</th>
            <th>Status</th>
            <th>Actions</th>
          </tr>
        </thead>
        <tbody>
          {signups.map(signup => (
            <tr key={signup.id}>
              <td>{new Date(signup.created_at).toLocaleDateString()}</td>
              <td>{signup.customer_name}</td>
              <td>{signup.business_name}</td>
              <td>{signup.tier_name}</td>
              <td>{signup.status}</td>
              <td>
                {signup.status === 'new' && (
                  <button onClick={() => handleSendPaymentLink(signup.id)}>

                )}
                {signup.status === 'payment_sent' && (

                    Mark Paid
                  </button>
                )}
              </td>
            </tr>
          ))}

// backend/routes/admin.js

app.get('/api/admin/signups', authenticateAdmin, async (req, res) => {
  const { status = 'new' } = req.query;
  
  const signups = await db.query(
    `SELECT sl.*, t.name as tier_name, t.price
     FROM signup_leads sl
     JOIN subscription_tiers t ON sl.tier_id = t.id
     WHERE sl.status = $1
     ORDER BY sl.created_at DESC`,
    [status]
  );
  
  res.json(signups.rows);
});

app.post('/api/admin/signups/:leadId/send-link', authenticateAdmin, async (req, res) => {
  const { leadId } = req.params;
  const { paymentLink } = req.body;
  
  const lead = await db.query(
    'SELECT * FROM signup_leads WHERE id = $1',
    [leadId]
  );
  
  // Log that link was sent
  await db.query(
    'INSERT INTO payment_links_sent (signup_lead_id, payment_link, recipient_email) VALUES ($1, $2, $3)',
    [leadId, paymentLink, lead.rows[0].customer_email]
  );
  
  // Send email to customer with payment link
  await sendPaymentLinkEmail(lead.rows[0].customer_email, paymentLink);
  
  // Update status
  await db.query(
    'UPDATE signup_leads SET status = $1, updated_at = NOW() WHERE id = $2',
    ['payment_sent', leadId]
  );
  
  res.json({ success: true });
});

app.post('/api/admin/signups/:leadId/mark-paid', authenticateAdmin, async (req, res) => {
  const { leadId } = req.params;
  
  const lead = await db.query(
    `SELECT sl.*, t.id as tier_id FROM signup_leads sl
     JOIN subscription_tiers t ON sl.tier_id = t.id
     WHERE sl.id = $1`,
    [leadId]
  );
  
  // Create business account
  const business = await db.query(
    `INSERT INTO businesses (business_name, email, phone)
     VALUES ($1, $2, $3)
     RETURNING id`,
    [lead.rows[0].business_name, lead.rows[0].customer_email, lead.rows[0].business_phone]
  );
  
  // Create subscription
  await db.query(
    `INSERT INTO subscriptions (business_id, tier_id, subscription_start, next_renewal_date, paid_at, payment_status)
     VALUES ($1, $2, NOW(), NOW() + INTERVAL '1 month', NOW(), 'paid')`,
    [business.rows[0].id, lead.rows[0].tier_id]
  );
  
  // Update signup lead status
  await db.query(
    'UPDATE signup_leads SET status = $1 WHERE id = $2',
    ['paid', leadId]
  );
  
  // Send activation email to customer
  await sendActivationEmail(lead.rows[0].customer_email, business.rows[0].id);
  
  res.json({ success: true });
});
