You are adding real-time notifications to the business owner. Right now when someone 
books, they don't know. They only find out when they check the dashboard. This kills trust.

CURRENT STATE:
- Backend: Node.js/Express
- Database: PostgreSQL
- Booking system: Complete (creates leads, sends customer emails)
- Slack/Email: Available

THE PROBLEM:
When a customer books an appointment:
1. Calendar updates ✓
2. Lead is created ✓
3. Customer gets confirmation email ✓
4. But the BUSINESS OWNER has NO IDEA it happened
5. They might miss the booking or see it hours later
6. Looks unprofessional

YOUR TASK:
Build a notification system that:
1. Sends immediate email to business owner when booking happens
2. (Optional) Sends Slack notification if they have Slack connected
3. Includes booking details, lead score, customer info
4. Tracks notification delivery
5. Business owner can configure what triggers notifications

IMPLEMENTATION REQUIREMENTS:

### STEP 1: Create Notification Settings Table

```sql
CREATE TABLE notification_settings (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  business_id UUID NOT NULL UNIQUE REFERENCES businesses(id) ON DELETE CASCADE,
  
  -- Email notifications
  notification_email VARCHAR(255), -- Where to send notifications
  email_enabled BOOLEAN DEFAULT true,
  email_on_booking BOOLEAN DEFAULT true,
  email_on_no_contact BOOLEAN DEFAULT false, -- If lead never confirms
  
  -- Slack notifications
  slack_webhook_url TEXT, -- Optional Slack webhook
  slack_enabled BOOLEAN DEFAULT false,
  slack_on_booking BOOLEAN DEFAULT true,
  slack_on_high_value BOOLEAN DEFAULT true, -- Only notify for quality_score > 75
  
  -- Quiet hours
  quiet_hours_start TIME,
  quiet_hours_end TIME,
  quiet_days TEXT[], -- ['Saturday', 'Sunday']
  
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE notification_logs (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  business_id UUID NOT NULL REFERENCES businesses(id),
  lead_id UUID NOT NULL REFERENCES leads(id),
  notification_type VARCHAR(50), -- 'booking', 'no_contact', 'follow_up'
  channel VARCHAR(50), -- 'email', 'slack'
  status VARCHAR(50), -- 'sent', 'failed', 'skipped'
  reason VARCHAR(255), -- Why it was skipped (e.g., 'quiet hours')
  sent_at TIMESTAMP,
  created_at TIMESTAMP DEFAULT NOW()
);
```

```javascript
// backend/services/notificationService.js

const { Resend } = require('resend');
const axios = require('axios');

const resend = new Resend(process.env.RESEND_API_KEY);

async function notifyBusinessOwner(businessId, leadData) {
  try {
    // 1. Get notification settings
    const settings = await db.query(
      'SELECT * FROM notification_settings WHERE business_id = $1',
      [businessId]
    );
    
    if (!settings.rows[0]) {
      return { success: false, reason: 'No notification settings' };
    }
    
    const notifSettings = settings.rows[0];
    
    // 2. Check if we should skip (quiet hours, weekend, etc.)
    if (shouldSkipNotification(notifSettings, leadData)) {
      await logNotification(businessId, leadData.id, 'email', 'skipped', 'quiet hours');
      return { success: true, skipped: true };
    }
    
    const results = [];
    
    // 3. Send email notification
    if (notifSettings.email_enabled && notifSettings.email_on_booking) {
      const emailResult = await sendOwnerEmailNotification(
        businessId,
        notifSettings.notification_email,
        leadData
      );
      results.push(emailResult);
    }
    
    // 4. Send Slack notification
    if (notifSettings.slack_enabled && notifSettings.slack_webhook_url) {
      if (notifSettings.slack_on_high_value && leadData.quality_score < 75) {
        // Skip this lead (low quality)
        await logNotification(businessId, leadData.id, 'slack', 'skipped', 'low quality score');
      } else {
        const slackResult = await sendSlackNotification(
          notifSettings.slack_webhook_url,
          leadData
        );
        results.push(slackResult);
      }
    }
    
    return { success: true, results };
    
  } catch (error) {
    console.error('Notification error:', error);
    return { success: false, error: error.message };
  }
}

async function sendSlackNotification(webhookUrl, leadData) {
  try {

            },
            {

              value: new Date(leadData.booked_datetime).toLocaleString(),

            },

    return { success: true, channel: 'slack' };
  } catch (error) {
    return { success: false, channel: 'slack', error: error.message };
  }
}

function shouldSkipNotification(settings, leadData) {
  // Check quiet hours
  if (settings.quiet_hours_start && settings.quiet_hours_end) {
    const now = new Date().toLocaleTimeString('en-US', { hour12: false });
    if (now >= settings.quiet_hours_start && now <= settings.quiet_hours_end) {
      return true;
    }
  }
  
  // Check quiet days
  if (settings.quiet_days && settings.quiet_days.length > 0) {
    const today = new Date().toLocaleDateString('en-US', { weekday: 'long' });
    if (settings.quiet_days.includes(today)) {
      return true;
    }
  }
  
  return false;
}

function getQualityLevel(score) {
  if (score > 75) return 'high';
  if (score > 50) return 'medium';
  return 'low';
}

async function logNotification(businessId, leadId, channel, status, reason = null) {
  await db.query(
    `INSERT INTO notification_logs 
     (business_id, lead_id, channel, status, reason, sent_at)
     VALUES ($1, $2, $3, $4, $5, NOW())`,
    [businessId, leadId, channel, status, reason]
  );
}
```

```javascript
app.post('/api/integrations/slack/connect', authenticateUser, async (req, res) => {
  const { businessId, slackWebhookUrl } = req.body;
  
  // Test webhook
  try {
    await axios.post(slackWebhookUrl, { text: '✅ Vernox AI Slack integration connected!' });

    return res.status(400).json({ error: 'Invalid Slack webhook URL' });

  // Save webhook
  await db.query(
    'UPDATE notification_settings SET slack_webhook_url = $1, slack_enabled = true WHERE business_id = $2',
    [slackWebhookUrl, businessId]
  );
  
  res.json({ success: true });
});

app.get('/api/integrations/slack/disconnect', authenticateUser, async (req, res) => {
  const { businessId } = req.query;

    'UPDATE notification_settings SET slack_webhook_url = NULL, slack_enabled = false WHERE business_id = $1',
    [businessId]
  );
```

### STEP 5: Notification Settings UI

```jsx
// frontend/pages/NotificationSettings.jsx

export default function NotificationSettings() {
  const [email, setEmail] = useState('');
  const [slackWebhook, setSlackWebhook] = useState('');
  const [quietHours, setQuietHours] = useState({ start: null, end: null });
  
  const handleSaveNotifications = async () => {
    await fetch(`/api/notification-settings/${businessId}`, {
      method: 'POST',
      body: JSON.stringify({
        notification_email: email,
        quiet_hours_start: quietHours.start,
        quiet_hours_end: quietHours.end
      })
    });
    alert('Notification settings saved!');
  };
  
  const handleConnectSlack = async () => {
    // Redirect to Slack OAuth or show webhook input
    const webhook = prompt('Paste your Slack webhook URL:');
    if (webhook) {
      const result = await fetch('/api/integrations/slack/connect', {
        method: 'POST',
        body: JSON.stringify({ businessId, slackWebhookUrl: webhook })
      });
      
      if (result.ok) {
        alert('Slack connected!');
        setSlackWebhook(webhook);
      } else {
        alert('Invalid webhook URL');
      }
    }
  };
  
  return (
    <div className="notification-settings">
      <h2>Notification Settings</h2>
      
      {/* Email Settings */}
      <section>
        <h3>Email Notifications</h3>
        <label>
          <input 
            type="email"
            value={email}
            onChange={(e) => setEmail(e.target.value)}
            placeholder="owner@business.com"
          />
          <span>Where to send booking notifications</span>
        </label>
      </section>
      
      {/* Slack Settings */}
      <section>
        <h3>Slack Notifications</h3>
        {slackWebhook ? (
          <>
            <p>✅ Slack connected</p>
            <button onClick={() => { /* disconnect */ }}>Disconnect</button>
          </>
        ) : (
          <button onClick={handleConnectSlack}>Connect Slack</button>
        )}
      </section>
      
      {/* Quiet Hours */}
      <section>
        <h3>Quiet Hours</h3>
        <label>
          <span>No notifications between:</span>
          <input 
            type="time"
            value={quietHours.start || ''}
            onChange={(e) => setQuietHours({ ...quietHours, start: e.target.value })}
          />
          <span>and</span>
          <input 
            type="time"
            value={quietHours.end || ''}
            onChange={(e) => setQuietHours({ ...quietHours, end: e.target.value })}
          />
        </label>
      </section>
      
      <button onClick={handleSaveNotifications}>Save Notification Settings</button>
    </div>
  );
}
```
